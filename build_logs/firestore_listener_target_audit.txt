   457	    List<ConversationSummary> mergedDocs() {
   458	      final byId = <String, ConversationSummary>{};
   459	      for (final docs in snapshotsByField.values) {
   460	        for (final doc in docs) {
   461	          final existing = byId[doc.id];
   462	          byId[doc.id] = existing == null ? doc : existing.mergeWith(doc);
   463	        }

   467	        ..sort((left, right) {
   468	          final leftDate = left.sortDate;
   469	          final rightDate = right.sortDate;
   470	          if (leftDate == null && rightDate == null) {
   471	            return left.id.compareTo(right.id);
   472	          }
   473	          if (leftDate == null) return 1;
   474	          if (rightDate == null) return -1;
   475	          return rightDate.compareTo(leftDate);
   476	        });
   477	      return docs;
